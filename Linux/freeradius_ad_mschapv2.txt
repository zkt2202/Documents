# Create CSR file based on key and configuration.

 -key /etc/freeradius/3.0/ssl/peap.example.biz.key\
 -out /etc/freeradius/3.0/ssl/peap.example.biz.csr\
 -extensions PEAP\
 -config /etc/freeradius/3.0/ssl/config.conf

# Test the CSR file
sudo openssl req -in /etc/freeradius/3.0/ssl/peap.example.biz.csr -noout -text

# Create server certificate
sudo openssl x509 -req -in /etc/freeradius/3.0/ssl/peap.example.biz.csr\
 -CA /etc/freeradius/3.0/ssl/example_CA.crt\
 -CAkey /etc/freeradius/3.0/ssl/example_CA.key\
 -CAcreateserial\
 -out /etc/freeradius/3.0/ssl/peap.example.biz.crt\
 -days 365 

# Run the SSL checking.

# Process script to requery AD database and return VLANID
# Install python ldap library
sudo python3 -m pip install ldap3

# Create script file
sudo mkdir /etc/freeradius/3.0/scripts
sudo bash -c "cat << EOF > /etc/freeradius/3.0/scripts/adQuery.py
import sys
import ldap3

def main():
    # return erroe when there is no agrument
    if len(sys.argv) < 1:
        return 5
    # set variable from agrument
    sam = sys.argv[1]
    # set variable
    bind_dn = 'cn=Searcher,cn=Users,dc=example,dc=biz'
    bind_passwd = '##########'
    base_dn = 'ou=users,dc=example,dc=biz'
    base_vlan_file = '/etc/freeradius/3.0/scripts/vlan_info.txt'
    # Process to connect and query data from AD with variable
    try:
        ad_servers = ldap3.Server('ad1.example.biz', port=636, use_ssl=True)
        ad_connect = ldap3.Connection(ad_servers, bind_dn, bind_passwd, auto_bind=True)
    except:
        return -1
    # Process to search department using login account (sam)
    try:
        ad_connect.search(
            base_dn,
            '(&(objectclass=user)(sAMAccountName=%s))' % sam,
            attributes=['department','description']
        )
    except:
        return -1
    # if there is no user name then return error
    if len(ad_connect.entries) == 0:
        return 5
    # get department from attribute department or description.
    result = str(ad_connect.entries[0].department.value) if ad_connect.entries[0].department else str(ad_connect.entries[0].description.value)
    # Read and add vlan information into dictionary
    data = {}
    with open(base_vlan_file) as f:
        for line in f:
            (k,v) = line.split(",")
            data[k] = int(v)
    # Get vlan id from data dictionary
    try:
        vlan_id = data.get(result)
    # if there is no match in dictionary, return default vlan_id
    except:
        vlan_id = '10'
    print('Tunnel-Type = VLAN')
    print('Tunnel-Medium-Type = 6')
    print('Tunnel-Private-Group-Id = ' + vlan_id)
    return 0

if __name__ == '__main__':
    main()
EOF"

# Do some setting which allow script to run.

    program = '/usr/bin/python3 /etc/freeradius/3.0/scripts/adQuery.py %{User-Name}'
    input_pairs = request
    output_pairs = reply
    shell_escape = yes
    timeout = 10
}
EOF"

# Next is creating a way to bypass from freeradius to ntlm_auth command which will return user identification.
sudo rm /etc/freeradius/3.0/mods-enabled/mschap
sudo bash -c "cat << EOF > /etc/freeradius/3.0/mods-enabled/mschap
mschap {
    use_mppe = yes
    require_encryption = yes
    require_strong = yes
    ntlm_auth = '/usr/bin/ntlm_auth --request-nt-key --allow-mschapv2 --username=%{%{Stripped-User-Name}:-%{%{User-Name}:-None}} --challenge=%{mschap:Challenge:-00} --nt-response=%{mschap:NT-Response:-00}'
}
EOF"
